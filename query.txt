CREATE TABLE user(
	id INT PRIMARY KEY AUTO_INCREMENT,
  	nome VARCHAR(50),
    ruolo VARCHAR(15) CHECK (ruolo IN ("produttore","raccoglitore","distributore")),
    email VARCHAR(100) UNIQUE,
    wallet_addr VARCHAR(256) UNIQUE,
    data_creazione_wallet TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    attivo BOOLEAN DEFAULT TRUE
);

CREATE TABLE log(
	id INT PRIMARY KEY AUTO_INCREMENT,
   	id_user INT NOT NULL,
    tipo_evento VARCHAR(100),
    ip_address VARCHAR(50) NOT NULL,
    log_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	
    FOREIGN KEY (id_user) REFERENCES user(id)

)

CREATE TABLE transaction_qrcode (
    tx_id VARCHAR(128) PRIMARY KEY,
    qrcode_img LONGTEXT NOT NULL   -- Base64 
)
